# --- MOTOR DE CÁLCULO ---
def poisson_pmf(k, lambda_val):
    return (math.pow(lambda_val, k) * math.exp(-lambda_val)) / math.factorial(k)

def analisar_partida(media_liga, time_casa, time_fora, odds):
    f_atq_casa = time_casa['gols_marcados'] / media_liga
    f_def_fora = time_fora['gols_sofridos'] / media_liga
    f_atq_fora = time_fora['gols_marcados'] / media_liga
    f_def_casa = time_casa['gols_sofridos'] / media_liga

    imp_atq_casa = 1.0 - time_casa['desf_ataque']
    imp_def_casa = 1.0 - time_casa['desf_defesa']
    imp_atq_fora = 1.0 - time_fora['desf_ataque']
    imp_def_fora = 1.0 - time_fora['desf_defesa']

    lambda_casa = max(0.2, (f_atq_casa * imp_atq_casa) * (f_def_fora / imp_def_fora) * media_liga * time_casa['fator_casa'])
    lambda_fora = max(0.2, (f_atq_fora * imp_atq_fora) * (f_def_casa / imp_def_casa) * media_liga)

    max_gols = 6
    matriz = np.zeros((max_gols, max_gols))
    for i in range(max_gols):
        for j in range(max_gols):
            matriz[i][j] = poisson_pmf(i, lambda_casa) * poisson_pmf(j, lambda_fora)

    prob_casa = np.sum(np.tril(matriz, -1))
    prob_empate = np.sum(np.diag(matriz))
    prob_fora = np.sum(np.triu(matriz, 1))

    prob_under25 = np.sum([matriz[i][j] for i in range(max_gols) for j in range(max_gols) if i + j < 2.5])
    prob_over25 = 1.0 - prob_under25
    prob_btts = np.sum(matriz[1:, 1:])

    odd_j_casa = 1 / prob_casa if prob_casa > 0 else 99
    odd_j_empate = 1 / prob_empate if prob_empate > 0 else 99
    odd_j_fora = 1 / prob_fora if prob_fora > 0 else 99

    ev_casa = (prob_casa * odds['casa']) - 1
    ev_empate = (prob_empate * odds['empate']) - 1
    ev_fora = (prob_fora * odds['fora']) - 1

    return {
        "xG": (round(lambda_casa, 2), round(lambda_fora, 2)),
        "probs": (prob_casa, prob_empate, prob_fora),
        "odds_justas": (odd_j_casa, odd_j_empate, odd_j_fora),
        "evs": (ev_casa, ev_empate, ev_fora),
        "over25": prob_over25,
        "btts": prob_btts
    }
